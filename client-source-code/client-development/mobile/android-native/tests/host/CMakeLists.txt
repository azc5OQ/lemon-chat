cmake_minimum_required(VERSION 3.22.1)
project(lemon_native_host_checks LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(JNI REQUIRED)
find_package(Threads REQUIRED)
get_filename_component(LEMON_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
get_filename_component(native_sources "${CMAKE_CURRENT_SOURCE_DIR}/../../app/src/main/cpp" ABSOLUTE)

add_library(lemonnative SHARED)
set_target_properties(lemonnative PROPERTIES C_VISIBILITY_PRESET hidden)
if(WIN32)
    set_target_properties(lemonnative PROPERTIES PREFIX "")
endif()
target_include_directories(lemonnative PRIVATE ${JNI_INCLUDE_DIRS})
target_link_libraries(lemonnative PRIVATE Threads::Threads)
include("${native_sources}/crypto.cmake")
include("${native_sources}/audio.cmake")
include("${native_sources}/transport.cmake")

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(lemonnative PRIVATE -Os -ffunction-sections -fdata-sections)
    if(APPLE)
        target_link_options(lemonnative PRIVATE -Wl,-dead_strip)
    else()
        target_link_options(lemonnative PRIVATE -Wl,--gc-sections)
    endif()
endif()
if(MINGW)
    target_link_options(lemonnative PRIVATE -static-libgcc -static-libstdc++)
endif()
