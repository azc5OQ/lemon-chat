if(LEMON_DESKTOP_SANITIZERS)
    if(NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang" OR WIN32)
        message(FATAL_ERROR "Application sanitizers currently require GCC or Clang on Linux/macOS.")
    endif()
    get_property(desktop_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
    foreach(desktop_target IN LISTS desktop_targets)
        get_target_property(desktop_type ${desktop_target} TYPE)
        if((desktop_type STREQUAL "STATIC_LIBRARY" OR desktop_type STREQUAL "EXECUTABLE")
            AND NOT desktop_target STREQUAL "lemon_desktop_image_runtime")
            target_compile_options(${desktop_target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        endif()
    endforeach()
    target_link_options(lemon_desktop_clib INTERFACE -fsanitize=address,undefined)
endif()
